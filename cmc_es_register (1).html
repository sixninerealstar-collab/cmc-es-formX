📡 ตั้งค่า Google Apps Script URL

วาง Web App URL จาก Google Apps Script (ตัวกลางส่งไป Discord)

📌 วิธีตั้งค่า — Google Apps Script (อัปเดต):
1. ไปที่ script.google.com → สร้าง Project ใหม่
2. วางโค้ดนี้ลงไป (แทนที่โค้ดเดิมทั้งหมด):
const DISCORD_WH = "วาง_DISCORD_WEBHOOK_URL_ที่นี่";

function doGet(e) {
  try {
    const payload = JSON.parse(e.parameter.payload);
    UrlFetchApp.fetch(DISCORD_WH, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify(payload)
    });
  } catch(err) { Logger.log(err); }
  return ContentService
    .createTextOutput("ok")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) { return doGet(e); } 3. Deploy → New deployment → Web app
